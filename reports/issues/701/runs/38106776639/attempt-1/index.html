export function isUniqueConstraintViolation(error) {
    const visited = new Set();
    let current = error;
    while (current && typeof current === 'object' && !visited.has(current)) {
        visited.add(current);
        const record = current;
        const code = record.code;
        const number = record.errno ?? record.number ?? record.errorNum;
        if (record.errCode === -6602 ||
            code === '23505' ||
            code === 'ER_DUP_ENTRY' ||
            code === 'SQLITE_CONSTRAINT' ||
            code === 'SQLITE_CONSTRAINT_UNIQUE' ||
            code === 'SQLITE_CONSTRAINT_PRIMARYKEY' ||
            number === 1 ||
            number === 1062 ||
            number === 2601 ||
            number === 2627) {
            return true;
        }
        current = record.cause ?? record.originalError;
    }
    return false;
}

import { I18nRuntime } from '@nocobase/i18n';
const LOCALE_STORAGE_KEY = 'nocobase.locale';
/** The language an application falls back to when its configuration names none. */
export const DEFAULT_LOCALE = 'en-US';
/**
 * Reads the visitor's stored language preference.
 *
 * Storage is the source of truth on the client, so the first frame renders in the right language without waiting for
 * the server. Access is guarded because a browser can refuse it outright in private modes.
 */
export function readStoredLocale() {
    try {
        return globalThis.localStorage?.getItem(LOCALE_STORAGE_KEY) ?? undefined;
    }
    catch {
        return undefined;
    }
}
export function writeStoredLocale(locale) {
    try {
        globalThis.localStorage?.setItem(LOCALE_STORAGE_KEY, locale);
    }
    catch {
        // A visitor who blocks storage still gets a working switch for this session.
    }
}
/**
 * Builds the i18n runtime for an application and loads the starting locale.
 *
 * Every contribution's resources for that one locale are fetched in parallel before this resolves, so the application
 * renders already translated instead of flashing keys behind a loading state. Other locales stay unfetched until the
 * visitor switches to one.
 */
export async function createAppI18nRuntime(options) {
    const applicationContribution = options.contributions.find((contribution) => contribution.source === 'application');
    const runtime = new I18nRuntime({
        defaultLocale: options.defaultLocale ?? DEFAULT_LOCALE,
        locales: options.locales,
        applicationNamespace: applicationContribution?.packageName,
    });
    for (const contribution of options.contributions) {
        if (contribution.source === 'application') {
            // Registering the application's namespace is also what settles which languages are on offer: its own locale
            // files are the list, and a plugin's only supply translations for languages already on it.
            runtime.registerApplicationNamespace(contribution.packageName, contribution.locales);
        }
        else {
            runtime.registerNamespace(contribution.packageName, contribution.locales);
        }
    }
    await runtime.init(options.initialLocale ?? readStoredLocale());
    return runtime;
}

        }

            throw cause;
        }
        finally {

        }
    }, [api, runtime, setLocale]);

        locale,
        locales,

    };

I18nRuntime 每个进程/浏览器标签页一个 i18next 实例；computeLocales 把应用命名空间声明的语言与 defaultLocale 求并集，默认语言始终在列；resolveLocale 把不支持的语言回退到默认。

export class I18nRuntime {
    options;
    registry = new I18nRegistry();
    i18n = createInstance();
    declaredLocales;
    locales;
    defaultLocale;
    initialized = false;
    localeLoads = new Map();
    constructor(options) {
        this.options = options;
        this.defaultLocale = options.defaultLocale;
        this.declaredLocales = options.locales;
        if (options.applicationNamespace) {
            this.registry.setApplicationNamespace(options.applicationNamespace);
        }
        this.locales = this.computeLocales();
    }
    /**
     * The languages on offer: whichever the application's own locale files declare, plus the default.
     *
     * A plugin's locale file supplies translations, not languages — an installed plugin that happens to ship `ja-JP`
     * must not put Japanese in the application's picker. The default is always present so a resolution has somewhere to
     * land even before any namespace has registered, and it is appended rather than prepended so changing it does not
     * reorder a language picker.
     */
    computeLocales() {
        const applicationNamespace = this.registry.getApplicationNamespace();
        const declared = this.declaredLocales ??
            (applicationNamespace
                ? this.registry.getNamespaceLocales(applicationNamespace)
                : []);
        return [...new Set([...declared, this.defaultLocale])];
    }
    getDefaultLocale() {
        return this.defaultLocale;
    }
    getLocales() {
        return this.locales;
    }

    /**
     * Initializes the instance for a locale, loading that locale's resources first so the first render is already
     * translated and no loading state is needed.
     */
    async init(locale = this.defaultLocale) {
        if (this.initialized) {
            await this.changeLanguage(locale);
            return this.i18n;
        }
        const resolved = this.resolveLocale(locale);
        this.initialized = true;
        await this.i18n.init({
            lng: resolved,
            // A chain, not one language: the application's default first, then English. A plugin that has not translated
            // the language in use falls back to the application's default, and to English when it lacks that one too — which
            // is what an application defaulting to Chinese and adding Spanish leaves a plugin facing.
            fallbackLng: [...new Set([this.defaultLocale, BASE_LOCALE])],
            defaultNS: this.registry.getApplicationNamespace() ?? BASE_NAMESPACE,
            // Keys nest, so `trigger.types.schedule` addresses a tree. Namespaces never travel inside the key: they are
            // passed through options, which keeps a colon in a key from being read as a namespace separator.
            keySeparator: '.',
            nsSeparator: false,
            // Resources are added by `ensureLocaleLoaded` rather than by a backend plugin, so there is nothing to defer and
            // `t()` is usable the moment `init` resolves.
            initAsync: false,
            interpolation: { escapeValue: false },
            react: {
                // Without this react-i18next keeps only the first namespace of the list it is given, which would drop the
                // fallback chain that `NamespaceScope` passes down and leave a plugin unable to reuse application wording.
                nsMode: 'fallback',

            ...this.options.initOptions,

# Goal

Put user-facing text behind a translation key correctly the first time: in the right file, under the right namespace, and in a way that survives a language switch.

# Scope

- Wire locale files into a package that has none yet.
- Add a string to an application or a plugin and render it translated, in the browser or on the server.
- Throw an error that callers see in their own language, and translate mail or jobs for their recipient.
- Add a language to a package that already has locale files.
- Reword a plugin's copy from the application, without editing the plugin.
- Translate a menu or breadcrumb title.

# Non-Goals

- Do not build a language picker. `useAppLocale()` exists; `app-template-default` already renders one in `client/layouts/components/language-switcher.tsx`.
- Do not add i18n machinery to an application. The plugin is registered by default, the client runtime is built by `createAppI18nRuntime`, and the server's by `I18nProvider`.
- Do not configure the list of languages. It is whatever the application's own `locales/index.ts` declares; configuration names only the default, `i18n.defaultLocale` (overridden by `APP_DEFAULT_LOCALE`). The `@nocobase/app-plugin-i18n` README covers that, `useAppLocale()`, the endpoints, and `useSyncServerLocale()`, which the template shell already calls to keep the server's language in step with the browser's.

# The one rule that decides everything

**A namespace is a package name, and the namespace in scope follows the render tree.**

An application's pages get the application's namespace; a plugin's pages get that plugin's, because the host wraps each contribution using the `packageName` it already records. So a component in its own package's routes translates with no namespace at all.

The exception is what most mistakes come from: a component a plugin exports for the application to render is in the _application's_ scope, not its own. It has to name its namespace.

# Copy and translation

All text a user can see goes through a translation key: titles, buttons, column headers, placeholders, `aria-label`, toasts and validation messages all count. Text hard-coded in a component cannot be translated, and changing its wording means changing code.

For the wording itself (which verb a button uses, how to write empty states and errors, how to put record names in quotes), see section C of [`../ui-guidelines.md`](../ui-guidelines.md); this document covers only how to write the code.

## Where copy lives

| File                             | Purpose                                                                                                                                              |
| -------------------------------- | ---------------------------------------------------------------------------------------------------------------------------------------------------- |
| `client/locales/system/en-US.ts` | The template's own English copy: the shell, the sign-in pages under `auth.*`, shared components and the homepage. Exports the type `SystemResource`  |
| `client/locales/system/zh-CN.ts` | The template's own Chinese copy; typed as `SystemResource`                                                                                           |
| `client/locales/en-US.ts`        | English: spreads `system/en-US.ts`, then adds the application's own groups; defines the structure of all the copy and exports the type `AppResource` |
| `client/locales/zh-CN.ts`        | Chinese: spreads `system/zh-CN.ts`, then adds the same groups; typed as `AppResource`, so its structure must match the English                       |
| `client/locales/index.ts`        | One dynamic `import()` per language, so the browser downloads only the language being shown                                                          |

Rules:

- Write the application's copy in `client/locales/en-US.ts` and `zh-CN.ts`, never in `client/locales/system/`. The system files belong to the template, and a template upgrade replaces them; keeping the two apart is what lets it do so without touching the application's copy.
- A group the application writes replaces the system group of the same name rather than merging with it. To add a key to a shared group such as `navigation` or `appearance`, spread the system group first: `navigation: { ...systemEnUS.navigation, projects: 'Projects' }`, and the same with `systemZhCN` in `zh-CN.ts`. Rewording a system key works the same way: write the key after the spread.
- Each new feature gets its own group (`projects: { … }`), which holds all of the feature's copy.
- Add menu titles to the existing `navigation` group (`navigation.projects`), and write `navigation: { title: 'navigation.projects' }` in the route (see [`page.md`](page.md)).
- Add the key to `en-US.ts` first, then add the same key to `zh-CN.ts`. A key missing from `zh-CN.ts` or extra in it (including a misspelled one) fails type checking.
- The sign-in pages' copy is the flat `auth.*` keys; the authentication blocks in `client/extensions/nocobase-auth-*/` take their text as props, so `client/pages/auth/` passes these translations in.
- **Keys passed to `t()` are not type-checked.** `t('projects.titel')` compiles, and the page shows `projects.titel` as is. `tests/logic/app-locale-coverage.test.ts` catches it; run it after every copy change (its limits are in ["Checks"](#checks)).
- The English built into a few shadcn/ui primitives, such as the visually hidden "Close" of `Sheet`, is translated too (guideline C1): pass the translation as a prop where the primitive takes one, and otherwise replace the literal in the primitive with a key. ["English built into primitives" in `shadcn.md`](shadcn.md#english-built-into-primitives) lists each case.

## Server-side text

API error messages, notifications and mail need translation too. The server translates them from `server/locales/`: `en-US.ts` states the wording and exports `AppServerResource`, and every other server locale is annotated with that type, so a key missing from one is a compile error. The template's server locales start empty — the server text that exists belongs to plugins — so add keys when the application's own endpoints start producing text. Name them by the same rules as client keys.

- The server resolves a request's language from the session's stored choice, then `Accept-Language`, then `i18n.defaultLocale`.
- Outbound content follows its recipient: mail, notifications and background work translate into the recipient's language, not the language of whoever triggered the work. A background job has no request to inherit a language from — resolve the recipient's locale explicitly and load it before translating. The `nocobase-app-plugin-i18n` Skill shows how server code translates, inside a request with `getRequestTranslator`, as an `AppI18nError` that the router's `onError` turns into an `ApiError` with a `localizedMessage`, and outside one with `getFixedT(namespace, locale)`.
- The frontend does not display an endpoint's `message` as is; it maps the status and the error's `reason` to its own copy (see [calling endpoints](frontend/references/api.md)).

## Keeping the server in step with the browser

`@nocobase/app-plugin-i18n` owns the language selector, persistence, and the server locale endpoints. The application shell calls `useSyncServerLocale()` so server messages use the browser's language when the server supports it; keep that call when changing the shell. If the server does not have that locale, the browser still switches and the server uses English; the language control shows an informational notice about the fallback.

## Which languages the application offers

The application's own locale files are the list. `client/locales/index.ts` decides what the browser offers and `server/locales/index.ts` what the server does, so adding a language means adding its file and loader entry on the side that needs it — there is no configured list to keep in step.

To add a language to the server, add its file annotated with `AppServerResource` (for example `server/locales/ja-JP.ts`) and its loader entry in `server/locales/index.ts`; the client side is described in [frontend copy and translations](frontend/references/i18n.md).

A plugin's locale file supplies translations for those languages, never a new one. A plugin shipping `ja-JP` to an application that does not offer Japanese puts nothing in the picker. Prefer keeping the client and server lists aligned when server-produced text needs the language. A client-only language is also valid: the interface switches to it, `PUT /api/i18n/locale` records English for the server, and the language control explains the fallback with an informational notice.
