            ...grant.origin,

                branchConstraints,

        resolveAll: async (input) => expand(await provider.resolveAll(input), input),
        resolve: async (input) => {
            const direct = await provider.resolve(input);
            const business = resources.definitionsList().length
                ? (await provider.resolveAll(input)).filter((grant) => grant.resource.type === 'resource')
                : [];
            const expanded = (await expand(business, input)).filter((grant) => grant.resource.type !== 'resource');

            if (input.resource.type === 'resource')

                grant.action === input.action);
        },

        onChange: provider.onChange?.bind(provider),
        scope: (identity) => composedGrants(provider.scope?.(identity) ?? provider, resources, constraints),
