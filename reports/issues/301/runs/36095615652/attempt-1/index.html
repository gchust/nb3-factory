 * Works without a runtime mounted — a focused test, or a host that has not set i18n up — where it returns each key's

export function useTranslation(ns, options) {
    const runtime = useOptionalI18nRuntime();
    const resolvedNamespaces = useMemo(() => (runtime && ns ? resolveNamespaces(runtime, ns) : ns), [runtime, ns]);
    const response = useI18nextTranslation(resolvedNamespaces, options);
    const underlyingT = response.t;
    const t = useCallback(((key, optionsOrDefaultValue, trailingOptions) => {
        if (!runtime) {
            return underlyingT(key, optionsOrDefaultValue, trailingOptions);
        }
        if (typeof optionsOrDefaultValue === 'string') {
            return underlyingT(key, optionsOrDefaultValue, resolveTranslationOptions(runtime, trailingOptions));
        }
        return underlyingT(key, resolveTranslationOptions(runtime, optionsOrDefaultValue));
    }), [runtime, underlyingT]);
    return useMemo(() => Object.assign([t, response.i18n, response.ready], {
        t,
        i18n: response.i18n,
        ready: response.ready,
    }), [response.i18n, response.ready, t]);
}
